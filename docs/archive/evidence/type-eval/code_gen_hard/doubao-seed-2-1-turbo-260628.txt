def tokenize(s):
    tokens, i, n = [], 0, len(s)
    while i < n:
        c = s[i]
        if c.isspace():
            i += 1
            continue
        if c in '()+-*/':
            tokens.append(('SYM', c))
            i += 1
        elif c in '=!<>':
            t = c + '=' if i+1 < n and s[i+1] == '=' else c
            tokens.append(('CMP', t))
            i += len(t)
        elif c in ('"', "'"):
            j = i + 1
            while j < n and s[j] != c: j += 1
            tokens.append(('STR', s[i+1:j]))
            i = j + 1
        elif c.isdigit() or c == '.':
            j = i
            while j < n and s[j] in '.0123456789': j += 1
            num = float(s[i:j])
            tokens.append(('NUM', int(num) if num.is_integer() else num))
            i = j
        elif c.isalpha() or c == '_':
            j = i
            while j < n and (s[j].isalnum() or s[j] == '_'): j += 1
            val = s[i:j]
            tokens.append(('KW' if val in ('and','or','not') else 'ID', val))
            i = j
        else:
            i += 1
    return tokens

def eval_cond(cond, customer):
    tokens = tokenize(cond)
    pos = [0]
    def peek():
        return tokens[pos[0]] if pos[0] < len(tokens) else None
    def parse_expr():
        left = parse_and()
        while peek() == ('KW', 'or'):
            pos[0] += 1
            if left: continue
            left = left or parse_and()
        return left
    def parse_and():
        left = parse_not()
        while peek() == ('KW', 'and'):
            pos[0] += 1
            if not left: continue
            left = left and parse_not()
        return left
    def parse_not():
        if peek() == ('KW', 'not'):
            pos[0] += 1
            return not parse_not()
        return parse_primary()
    def parse_primary():
        if peek() == ('SYM', '('):
            pos[0] += 1
            res = parse_expr()
            if peek() == ('SYM', ')'): pos[0] += 1
            return res
        return parse_cmp()
    def parse_cmp():
        left = parse_operand()
        tok = peek()
        if not tok or tok[0] != 'CMP': return left
        op = tok[1]
        pos[0] += 1
        right = parse_operand()
        try:
            if op == '==': return left == right
            if op == '!=': return left != right
            if op == '>': return left > right
            if op == '>=': return left >= right
            if op == '<': return left < right
            if op == '<=': return left <= right
        except TypeError:
            return False
    def parse_operand():
        tok = peek()
        pos[0] += 1
        if tok[0] == 'ID':
            if tok[1] not in customer: raise KeyError()
            return customer[tok[1]]
        return tok[1]
    try:
        return bool(parse_expr())
    except:
        return False

def eval_arith(expr, customer):
    tokens = tokenize(expr)
    pos = [0]
    def peek():
        return tokens[pos[0]] if pos[0] < len(tokens) else None
    def parse_expr():
        left = parse_term()
        while peek() and peek()[0] == 'SYM' and peek()[1] in '+-':
            op = peek()[1]
            pos[0] += 1
            left += parse_term() if op == '+' else -parse_term()
        return left
    def parse_term():
        left = parse_factor()
        while peek() and peek()[0] == 'SYM' and peek()[1] in '*/':
            op = peek()[1]
            pos[0] += 1
            right = parse_factor()
            left = left * right if op == '*' else left / right
        return left
    def parse_factor():
        tok = peek()
        if tok and tok[0] == 'SYM' and tok[1] in '+-':
            pos[0] += 1
            return -parse_factor() if tok[1] == '-' else parse_factor()
        return parse_primary()
    def parse_primary():
        tok = peek()
        pos[0] += 1
        if tok[0] == 'NUM': return tok[1]
        if tok[0] == 'ID': return customer.get(tok[1], 0)
        if tok[0] == 'SYM' and tok[1] == '(':
            res = parse_expr()
            if peek() == ('SYM', ')'): pos[0] += 1
            return res
        return 0
    return int(parse_expr() // 1)

def evaluate_rules(customer, rules):
    for rule in rules:
        if eval_cond(rule['cond'], customer):
            p = rule['action']['points']
            val = int(p // 1) if isinstance(p, (int, float)) else eval_arith(p, customer)
            return {'points': val}
    return None
